    def __init__(
            self,
            pretrained=True,
            num_frames=1,
            input_norm=True,
            mid_dim=64,
            siu_groups=4,  # 参数保留，防止外部传参报错，内部不再使用
            hmu_groups=6,  # 参数保留，防止外部传参报错，内部不再使用
            model_name="swin_base_patch4_window12_384",  # COD 常用 384 输入分辨率
            **kwargs
    ):
        super().__init__()

        # 使用 timm 构建 Swin-Base
        self.encoder = timm.create_model(
            model_name=model_name,
            features_only=True,
            out_indices=(0, 1, 2, 3),  # Swin 默认输出 4 个尺度的特征
            pretrained=pretrained
        )

        # Swin-Base 的通道数通常为 [128, 256, 512, 1024]
        self.embed_dims = [128, 256, 512, 1024]

        # 🚀 消融实验设置：完全移除多尺度 SIU 和时序/记忆 HMU，只保留基础的特征转换层

        # --- 第 5 层 (对应 Swin 的最后一层，通道 1024, stride 32) ---
        self.tra_5 = SimpleASPP(self.embed_dims[3], out_dim=mid_dim)

        # --- 第 4 层 (通道 512, stride 16) ---
        self.tra_4 = ConvBNReLU(self.embed_dims[2], mid_dim, 3, 1, 1)

        # --- 第 3 层 (通道 256, stride 8) ---
        self.tra_3 = ConvBNReLU(self.embed_dims[1], mid_dim, 3, 1, 1)

        # --- 第 2 层 (通道 128, stride 4) ---
        self.tra_2 = ConvBNReLU(self.embed_dims[0], mid_dim, 3, 1, 1)

        # --- 第 1 层 (从 stride 4 上采样到 stride 2) ---
        self.tra_1 = nn.Sequential(
            nn.Upsample(scale_factor=2, mode="bilinear", align_corners=False),
            ConvBNReLU(mid_dim, mid_dim, 3, 1, 1)
        )

        self.normalizer = PixelNormalizer() if input_norm else nn.Identity()
        self.predictor = nn.Sequential(
            nn.Upsample(scale_factor=2, mode="bilinear", align_corners=False),  # stride 2 恢复到 stride 1
            ConvBNReLU(mid_dim, 32, 3, 1, 1),
            nn.Conv2d(32, 1, 1),
        )

    def normalize_encoder(self, x):
        x = self.normalizer(x)
        features = self.encoder(x)

        formatted_features = []
        for f in features:
            # Swin Transformer 提取的特征通常是 NHWC 格式，例如 [16, 12, 12, 1024]
            # 我们需要把它调整为 PyTorch 标准的 NCHW 格式: [16, 1024, 12, 12]
            if f.dim() == 4 and f.shape[-1] in self.embed_dims:
                f = f.permute(0, 3, 1, 2).contiguous()
            formatted_features.append(f)

        c2, c3, c4, c5 = formatted_features
        return c2, c3, c4, c5

    def body(self, data):
        # 依然提取多尺度特征，但消融实验中只用中尺度 (m)
        m_trans_feats = self.normalize_encoder(data["image_m"])

        # --- 第5层消融 ---
        # 原代码：lms = self.siu_5(l=l, m=m, s=s); x = self.hmu_5(lms)
        # 消融后：直接使用中尺度特征，跳过 SIU 和 HMU
        x = self.tra_5(m_trans_feats[3])

        # --- 第4层消融 ---
        m_feat_4 = self.tra_4(m_trans_feats[2])
        x = m_feat_4 + resize_to(x, tgt_hw=m_feat_4.shape[-2:])

        # --- 第3层及以下以此类推 ---
        m_feat_3 = self.tra_3(m_trans_feats[1])
        x = m_feat_3 + resize_to(x, tgt_hw=m_feat_3.shape[-2:])

        m_feat_2 = self.tra_2(m_trans_feats[0])
        x = m_feat_2 + resize_to(x, tgt_hw=m_feat_2.shape[-2:])

        x = self.tra_1(x)
        return self.predictor(x)

    def get_grouped_params(self):
        # 针对 Swin 调整冻结策略
        param_groups = {"pretrained": [], "fixed": [], "retrained": []}
        for name, param in self.named_parameters():
            # 冻结 Swin 第一层的 patch embedding（防止前期训练抖动过大）
            if name.startswith("encoder.patch_embed."):
                param.requires_grad = False
                param_groups["fixed"].append(param)
            elif name.startswith("encoder."):
                param_groups["pretrained"].append(param)
            else:
                if "clip." in name:
                    param.requires_grad = False
                    param_groups["fixed"].append(param)
                else:
                    param_groups["retrained"].append(param)

        LOGGER.info(
            f"Parameter Groups:{{"
            f"Pretrained: {len(param_groups['pretrained'])}, "
            f"Fixed: {len(param_groups['fixed'])}, "
            f"ReTrained: {len(param_groups['retrained'])}}}"
        )
        return param_groups
